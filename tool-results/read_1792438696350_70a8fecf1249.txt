   504→   504→export async function getSiteData(): Promise<SiteData> {
   505→   505→  try {
   506→   506→    const [serviceRows, testimonialRows, faqRows, officeRows] = await Promise.all([
   507→   507→      db.service.findMany({ where: { visible: true }, orderBy: [{ order: "asc" }, { createdAt: "asc" }] }),
   508→   508→      db.testimonial.findMany({ where: { visible: true }, orderBy: [{ order: "asc" }, { createdAt: "asc" }] }),
   509→   509→      db.faq.findMany({ where: { visible: true }, orderBy: [{ order: "asc" }, { createdAt: "asc" }] }),
   510→   510→      db.office.findMany({ orderBy: [{ order: "asc" }, { createdAt: "asc" }] }),
   511→   511→    ]);
   512→   512→
   513→   513→    const services: ServiceItem[] = serviceRows.map((s) => {
   514→   514→      let items: string[] = [];
   515→   515→      try {
   516→   516→        items = JSON.parse(s.features);
   517→   517→      } catch {
   518→   518→        items = [];
   519→   519→      }
   520→   520→      return {
   521→   521→        slug: s.slug,
   522→   522→        title: s.title,
   523→   523→        desc: s.desc,
   524→   524→        price: s.price,
   525→   525→        items,
   526→   526→        icon: s.icon,
   527→   527→        category: s.category,
   528→   528→        featured: s.featured,
   529→   529→      };
   530→   530→    });
   531→   531→
   532→   532→    return {
   533→   533→      settings: DEFAULT_SETTINGS,
   534→   534→      services: services.length > 0 ? services : DEFAULT_SERVICES,
   535→   535→      testimonials:
   536→   536→        testimonialRows.length > 0
   537→   537→          ? testimonialRows.map((t) => ({ name: t.name, role: t.role, text: t.text, rating: t.rating }))
   538→   538→          : DEFAULT_TESTIMONIALS,
   539→   539→      faqs: faqRows.length > 0 ? faqRows.map((f) => ({ q: f.question, a: f.answer })) : DEFAULT_FAQS,
   540→   540→      offices: officeRows.length > 0
   541→   541→        ? officeRows.map((o) => ({ type: o.type, label: o.label, address: o.address }))
   542→   542→        : DEFAULT_OFFICES,
   543→   543→    };
   544→   544→  } catch (e) {
   545→   545→    console.error("getSiteData fallback ke data default:", e);
   546→   546→    return {
   547→   547→      settings: DEFAULT_SETTINGS,
   548→   548→      services: DEFAULT_SERVICES,
   549→   549→      testimonials: DEFAULT_TESTIMONIALS,
   550→   550→      faqs: DEFAULT_FAQS,
   551→   551→      offices: DEFAULT_OFFICES,
   552→   552→    };
   553→   553→  }
   554→   554→}

   556→   556→/** Format Rupiah ringkas gaya Indonesia: Rp 7,5 jt / Rp 850 rb */
   557→   557→export function fmtRp(v: number): string {
   558→   558→  const n = v >= 1_000_000_000 ? v / 1_000_000_000 : v >= 1_000_000 ? v / 1_000_000 : v / 1_000;
   559→   559→  const s = n.toLocaleString("id-ID", { maximumFractionDigits: 1 });
   560→   560→  if (v >= 1_000_000_000) return `Rp ${s} M`;
   561→   561→  if (v >= 1_000_000) return `Rp ${s} jt`;
   562→   562→  if (v >= 1_000) return `Rp ${s} rb`;
   563→   563→  return `Rp ${v.toLocaleString("id-ID")}`;
   564→   564→}
