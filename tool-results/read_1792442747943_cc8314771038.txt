   289→export default function OsExplorer() {
   290→  const [data, setData] = useState<GunaraApiResponse | null>(null);
   291→  const [status, setStatus] = useState<LoadStatus>("loading");
   292→  const [query, setQuery] = useState("");
   293→  const [tierFilter, setTierFilter] = useState<string | null>(null);
   294→  const [domainFilter, setDomainFilter] = useState<number | null>(null);
   295→  const [selectedDoc, setSelectedDoc] = useState<PgaDocData | null>(null);
   296→  const [addedDocs, setAddedDocs] = useState<Set<string>>(() => new Set());
   297→
   298→  const loadData = useCallback(async () => {
   299→    setStatus("loading");
   300→    try {
   301→      const res = await fetch("/api/gunara", { cache: "no-store" });
   302→      if (!res.ok) throw new Error(`HTTP ${res.status}`);
   303→      const json = (await res.json()) as GunaraApiResponse;
   304→      setData(json);
   305→      setStatus("ready");
   306→    } catch {
   307→      setData(null);
   308→      setStatus("error");
   309→    }
   310→  }, []);
   311→
   312→  useEffect(() => {
   313→    void loadData();
   314→  }, [loadData]);
   315→
   316→  // Sinkronkan status "sudah di penawaran" — hanya di client effect (SSR-safe).
   317→  useEffect(() => {
   318→    setAddedDocs(new Set(readDocCodes()));
   319→  }, []);
   320→
   321→  const domains = data?.domains ?? [];
   322→  const docs = data?.docs ?? [];
   323→  const packages = data?.packages ?? [];
   324→
   325→  const domainNameById = useMemo(() => {
   326→    const map = new Map<number, string>();
   327→    for (const d of domains) map.set(d.number, d.name);
   328→    return map;
   329→  }, [domains]);
   330→
   331→  const domainCounts = useMemo(() => {
   332→    const map = new Map<number, number>();
   333→    for (const doc of docs) {
   334→      map.set(doc.domainNumber, (map.get(doc.domainNumber) ?? 0) + 1);
   335→    }
   336→    return map;
   337→  }, [docs]);
   338→
   339→  const filteredDocs = useMemo(() => {
   340→    const q = query.trim().toLowerCase();
   341→    return docs.filter((doc) => {
   342→      if (tierFilter && doc.tier !== tierFilter) return false;
   343→      if (domainFilter !== null && doc.domainNumber !== domainFilter) return false;
   344→      if (!q) return true;
   345→      return (
   346→        doc.code.toLowerCase().includes(q) ||
   347→        doc.title.toLowerCase().includes(q) ||
   348→        doc.owner.toLowerCase().includes(q)
   349→      );
   350→    });
   351→  }, [docs, query, tierFilter, domainFilter]);
   352→
   353→  const handleAddDoc = useCallback((doc: PgaDocData) => {
   354→    addDocCode(doc.code);
   355→    setAddedDocs((prev) => new Set(prev).add(doc.code));
   356→    toast({
   357→      title: "Dokumen ditambahkan",
   358→      description: `${doc.code} — ${doc.title} masuk ke draft penawaran Anda.`,
   359→    });
   360→  }, []);
   361→
   362→  const resetFilters = useCallback(() => {
   363→    setQuery("");
   364→    setTierFilter(null);
   365→    setDomainFilter(null);
   366→  }, []);
   367→
   368→  const stats = [
   369→    { value: data ? String(data.docs.length) : "165", label: "Dokumen Standar" },
   370→    { value: data ? String(data.domains.length) : "20", label: "Domain Kepemimpinan" },
   371→    { value: String(TIER_KEYS.length), label: "Tier Prioritas" },
   372→    { value: data ? String(data.packages.length) : "5", label: "Paket Standar" },
   373→  ];
   374→
   375→  const selectedDomainName = selectedDoc
   376→    ? (domainNameById.get(selectedDoc.domainNumber) ?? `Domain ${selectedDoc.domainNumber}`)
   377→    : "";
   378→
   379→  return (
   380→    <div className="relative min-h-screen bg-night text-foreground">
   381→      {/* Dekorasi latar */}
   382→      <div
   383→        aria-hidden="true"
   384→        className="pointer-events-none absolute inset-x-0 top-0 h-[620px] starfield opacity-40"
   385→      />
   386→      <div
   387→        aria-hidden="true"
   388→        className="pointer-events-none absolute inset-x-0 top-0 h-[620px] grid-lines"
   389→      />
   390→
   391→      <main className="relative">
   392→        {status === "loading" && <ExplorerSkeleton />}
   393→        {status === "error" && <ExplorerError onRetry={() => void loadData()} />}
   394→
   395→        {status === "ready" && (
   396→          <>
   397→            {/* ═══════════ HEADER ═══════════ */}
   398→            <header className="section-padding pt-24 sm:pt-28">
